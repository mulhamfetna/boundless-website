{{- /* Bundles the Boundless stylesheets and emits one fingerprinted . Why this exists rather than @import inside custom.css: Hugo's resources.Concat joins files byte-for-byte and does not resolve CSS @import, so imports silently never load. Concatenating explicitly here keeps the source split into focused files (tokens, RTL, fonts) while still shipping a single request. This link renders after the theme's own bundle, so equal-specificity rules here win — which is what the RTL corrections need. Order matters for readability, not for var() resolution: custom properties are resolved at computed-value time, so a var() used in the theme bundle still resolves against tokens declared here. */ -}} {{- $sheets := slice -}} {{- with resources.Get "css/fonts.css" }}{{ $sheets = $sheets | append . }}{{ end -}} {{- with resources.Get "css/tokens.css" }}{{ $sheets = $sheets | append . }}{{ end -}} {{- with resources.Get "css/rtl.css" }}{{ $sheets = $sheets | append . }}{{ end -}} {{- if $sheets -}} {{- $bundle := $sheets | resources.Concat "css/boundless.bundle.css" | resources.Minify | resources.Fingerprint "sha256" -}} {{- end -}}
